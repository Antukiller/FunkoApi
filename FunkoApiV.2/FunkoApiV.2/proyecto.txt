using System.Text.Json;
using FunkoApiMongo.Config;
using FunkoApiV._2.Infrastructure.Interfaces;
using Microsoft.Extensions.Caching.Distributed;
using Microsoft.Extensions.Options;

namespace FunkoApiMongo.Cache;

public class RedisService(IDistributedCache cache, IOptions<CacheConfig> option) : ICacheService, ISingletonService {
    private readonly CacheConfig _config = option.Value;
    private readonly SemaphoreSlim _factory = new(1, 1);
    
    private readonly JsonSerializerOptions _jsonOptions = new() {
        PropertyNameCaseInsensitive = true
    };
    
    public async Task<T?> GetAsync<T>(string cacheKey, CancellationToken token = default) {
        var data = await cache.GetStringAsync(cacheKey, token);
        return string.IsNullOrEmpty(data) ? default : JsonSerializer.Deserialize<T>(data, _jsonOptions);
    }

    public async Task SetAsync<T>(string cacheKey, T value, TimeSpan? expiration = null, CancellationToken token = default) {
        var entryOptions = new DistributedCacheEntryOptions
        {
            AbsoluteExpirationRelativeToNow = expiration ?? _config.DefaultExpiration,
            SlidingExpiration = _config.SlidingExpiration
        };
        await cache.SetStringAsync(cacheKey, JsonSerializer.Serialize(value, _jsonOptions), entryOptions, token);
    }

    public async Task RemoveAsync(string cacheKey, CancellationToken token = default) {
        await cache.RemoveAsync(cacheKey, token);
    }

    public async Task<bool> ExistsAsync(string cacheKey, CancellationToken token = default) {
        var data = await cache.GetStringAsync(cacheKey, token);
        return !string.IsNullOrEmpty(data);
    }

    public async Task<T?> GetOrSetAsync<T>(string cacheKey, Func<Task<T?>> factory, CancellationToken token = default, TimeSpan? expiration = null) {
        var cached = await GetAsync<T>(cacheKey, token);
        if (cached is not null) return cached;

        await _factory.WaitAsync(token);
        try {
            cached = await GetAsync<T>(cacheKey, token);
            if (cached is not null) return cached;

            var value = await factory();
            if (value is not null)
                await SetAsync(cacheKey, value, expiration, token);
            return value;
        } finally {
            _factory.Release();
        }
    }
}

using System.Security.Claims;
using FunkoApiV._2.Dto.Auth;
using FunkoApiV._2.Error;      
using FunkoApiV._2.Service;
using Microsoft.AspNetCore.Authorization;
using Microsoft.AspNetCore.Mvc;

namespace FunkoApiV._2.Controller.Auth;

/// <summary>
/// Controller de autenticación: login, signup y perfil.
/// </summary>
[ApiController]
[Route("api/[controller]")]
public class AuthController(IAuthService authService) : ControllerBase {
    /// <summary>
    /// Inicia sesión con email y contraseña.
    /// </summary>
    [HttpPost("login")]
    public async Task<ActionResult<AuthResponseDto>> Login([FromBody] LoginDto dto) {
        var result = await authService.Login(dto);
        return result.IsSuccess
            ? Ok(result.Value)
            : result.Error.TopHttpResult<AuthResponseDto>();
    }

    /// <summary>
    /// Registra un nuevo usuario.
    /// </summary>
    [HttpPost("signup")]
    public async Task<ActionResult<AuthResponseDto>> Signup([FromBody] SignupDto dto) {
        var result = await authService.Signup(dto);
        return result.IsSuccess
            ? Ok(result.Value)
            : result.Error.TopHttpResult<AuthResponseDto>();
    }

    /// <summary>
    /// Obtiene la información del usuario autenticado.
    /// </summary>
    [Authorize]
    [HttpGet("me")]
    public async Task<ActionResult<UserDto>> GetMe() {
        var userIdClaim = User.FindFirst(ClaimTypes.NameIdentifier)?.Value;
        
        if (!long.TryParse(userIdClaim, out long userId)) {
            return Unauthorized(new { message = "Token inválido o usuario no identificado" });
        }

        var result = await authService.GetMe(userId);
        return result.IsSuccess
            ? Ok(result.Value)
            : result.Error.TopHttpResult<UserDto>();
    }
}

using System.Text.Json;
using System.Text.Json.Serialization;
using MongoDB.Bson;

namespace FunkoApiV._2.Converters;

public class ObjectIdJsonConverter : JsonConverter<ObjectId> {
    
    public override ObjectId Read(ref Utf8JsonReader reader, Type typeToConvert, JsonSerializerOptions options) {
        var value = reader.GetString();
        return string.IsNullOrEmpty(value) ? ObjectId.Empty : new ObjectId(value);
    }

    public override void Write(Utf8JsonWriter writer, ObjectId value, JsonSerializerOptions options) {
        writer.WriteStringValue(value.ToString());
    }
}

    public int TotalCount { get; init; }

    public int PageSize { get; init; }

using Microsoft.AspNetCore.Authentication;
using Microsoft.AspNetCore.Authentication.JwtBearer;
using Microsoft.AspNetCore.Authorization;
using Microsoft.AspNetCore.Identity;
using Microsoft.IdentityModel.Tokens;
using Serilog;

namespace FunkoApiV._2.Infrastructure;

public static class IdentityConfig
{
    /// <summary>
    /// Registra Identity, JWT Bearer y políticas de autorización.
    /// </summary>
    /// <param name="services">Colección de servicios.</param>
    /// <param name="configuration">Configuración de la app.</param>
    /// <returns>IServiceCollection para encadenar.</returns>
    public static IServiceCollection AddIdentityConfiguration(
        this IServiceCollection services,
        IConfiguration configuration)
    {
        Log.Information("Configurando Identity y JWT...");

        // ── JWT Settings ─────────────────────────────────────────────────────
        var jwtSection = configuration.GetSection(JwtSettings.Section);
        services.Configure<JwtSettings>(jwtSection);

        var jwtSettings = jwtSection.Get<JwtSettings>() ?? new JwtSettings();

        if (string.IsNullOrWhiteSpace(jwtSettings.Key))
            throw new InvalidOperationException("JWT Key no configurada en appsettings.json");

        // ── Identity ─────────────────────────────────────────────────────────
        // AddIdentityCore: Identity sin UI (no registra la cookie Identity.Application),
        // el mecanismo oficial para APIs.
        services.AddIdentityCore<AppUser>(options =>
        {
            // Password
            options.Password.RequiredLength = 8;
            options.Password.RequireDigit = true;
            options.Password.RequireLowercase = true;
            options.Password.RequireUppercase = true;
            options.Password.RequireNonAlphanumeric = true;

            // Lockout
            options.Lockout.DefaultLockoutTimeSpan = TimeSpan.FromMinutes(5);
            options.Lockout.MaxFailedAccessAttempts = 5;

            // User
            options.User.RequireUniqueEmail = true;
        })
        .AddRoles<IdentityRole<long>>()
        .AddSignInManager()
        .AddEntityFrameworkStores<AppDbContext>()  
        .AddDefaultTokenProviders();

        // ── JWT Authentication ───────────────────────────────────────────────
        services.AddAuthentication(options =>
        {
            options.DefaultScheme = JwtBearerDefaults.AuthenticationScheme;
            options.DefaultAuthenticateScheme = JwtBearerDefaults.AuthenticationScheme;
            options.DefaultChallengeScheme = JwtBearerDefaults.AuthenticationScheme;
            options.DefaultForbidScheme = JwtBearerDefaults.AuthenticationScheme;
        })
        .AddJwtBearer(options =>
        {
            options.TokenValidationParameters = new TokenValidationParameters
            {
                ValidateIssuer = true,
                ValidateAudience = true,
                ValidateLifetime = true,
                ValidateIssuerSigningKey = true,
                ValidIssuer = jwtSettings.Issuer,
                ValidAudience = jwtSettings.Audience,
                IssuerSigningKey = new SymmetricSecurityKey(
                    Encoding.UTF8.GetBytes(jwtSettings.Key)),
                ClockSkew = TimeSpan.Zero
            };
        });

        // ── Authorization Policies ───────────────────────────────────────────
        services.AddAuthorizationBuilder()
            .AddPolicy("RequireAdmin", policy =>
                policy.RequireRole("ADMIN").Build())
            .AddPolicy("RequireUser", policy =>
                policy.RequireRole("USER", "ADMIN").Build());

        Log.Information("Identity y JWT configurados correctamente.");
        return services;
    }

using FunkoApiV._2.Dto;

using FunkoApiV._2.Models;

namespace FunkoApiV._2.Repository;

public class  CategoriaRepository(AppDbContext context) : ICategoriaRepository, IScopedService {
    public async Task<(IReadOnlyList<Categoria> Items, int Total)> GetPagedAsync(CategoriaFilterRequest filter) {
        IQueryable<Categoria> query = context.Categorias.AsNoTracking();

        if (filter.IncludeDeleted)
            query = query.IgnoreQueryFilters();

        if (!string.IsNullOrWhiteSpace(filter.Nombre))
            query = query.Where(c => EF.Functions.ILike(c.Nombre, $"%{filter.Nombre}%"));

        var total = await query.CountAsync();
        var items = await ApplySort(query, filter.SortBy, filter.Direction)
            .Skip(filter.Page * filter.Size)
            .Take(filter.Size)
            .ToListAsync();

        return (items, total);
    }

    public async Task<Result<Categoria, DomainError>> GetById(string id) {
        var categoria = await context.Categorias
            .AsNoTracking()
            .FirstOrDefaultAsync(c => c.Id == id && !c.IsDeleted);
        return categoria is not null
            ? Result.Success<Categoria, DomainError>(categoria)
            : Result.Failure<Categoria, DomainError>(NotFound.FromId(id, "Categoria"));
    }

using System.IdentityModel.Tokens.Jwt;
using System.Security.Claims;
using System.Text;
using CSharpFunctionalExtensions;
using FunkoApiV._2.Config;
using FunkoApiV._2.Dto.Auth;
using FunkoApiV._2.Entity;
using FunkoApiV._2.Error;

using Microsoft.IdentityModel.Tokens;

namespace FunkoApiV._2.Service;

public class AuthService(
    AppDbContext context, 
    UserManager<AppUser> userManager, 
    SignInManager<AppUser> signUser, 
    IOptions<JwtSettings> options) : IAuthService, IScopedService {
    
    public async Task<Result<AuthResponseDto, DomainError>> Login(LoginDto loginDto) {
        Log.Information("Email: {Email} logeandose", loginDto.Email);
        var user = await userManager.FindByEmailAsync(loginDto.Email);
        if (user == null) {
            Log.Warning("Login fallido: Usuario con emais {Email} no encontrado", loginDto.Email);
            return Result.Failure<AuthResponseDto, DomainError>(new UnauthorizedError("Error de logging"));
        }
        var res = await signUser.CheckPasswordSignInAsync(user, loginDto.Password, false);
        if (!res.Succeeded) {
            Log.Warning("Login fallido: Contraseña incorrecta");
            return Result.Failure<AuthResponseDto, DomainError>(new UnauthorizedError("Contraseña incorrecta"));
        }
        var role = (await userManager.GetRolesAsync(user)).FirstOrDefault() ?? "USER";
        var token = GenerateJwtToken(user, role);

        Log.Information("Login exitoso para {Email} con rol {Role}", loginDto.Email, role);
        return Result.Success<AuthResponseDto, DomainError>(token);
    }

    public async Task<Result<AuthResponseDto, DomainError>> Signup(SignupDto signupDto) {
        Log.Information("Intento de registro para {Email}", signupDto.Email);

        var existingUser = await userManager.FindByEmailAsync(signupDto.Email);
        if (existingUser is not null) {
            Log.Warning("Registro fallido: {Email} ya existe", signupDto.Email);
            return Result.Failure<AuthResponseDto, DomainError>(
                new Conflict($"El usuario con email {signupDto.Email} ya existe"));
        }
        var user = new AppUser {
            UserName = signupDto.Email,
            Email = signupDto.Email,
            Nombre = signupDto.Nombre,
            Apellido = signupDto.Apellido,
            Createdt = DateTime.UtcNow
        };

        var result = await userManager.CreateAsync(user, signupDto.Password);
        if (!result.Succeeded) {
            var errors = string.Join(", ", result.Errors.Select(e => e.Description));
            Log.Warning("Registro fallido para {Email}: {Errors}", signupDto.Email, errors);
            return Result.Failure<AuthResponseDto, DomainError>(
                new BusinessRuleError($"Error al crear el usuario: {errors}"));
        }

        await userManager.AddToRoleAsync(user, "USER");

        var role = "USER";
        var token = GenerateJwtToken(user, role);

        Log.Information("Registro exitoso para {Email} con rol {Role}", signupDto.Email, role);
        return Result.Success<AuthResponseDto, DomainError>(token);
    }

    public async Task<Result<UserDto, DomainError>> GetMe(long userId) {
        Log.Debug("Consultando usuario {UserId}", userId);

        var user = await context.Users.FindAsync(userId);
        if (user is null) {
            return Result.Failure<UserDto, DomainError>(
                NotFound.FromId(userId.ToString(), "Usuario"));
        }
        var role = (await userManager.GetRolesAsync(user)).FirstOrDefault() ?? "USER";
        return Result.Success<UserDto, DomainError>(
            new UserDto(user.Id, user.Email ?? "", user.Nombre, user.Apellido, role));
    }
    
    private AuthResponseDto GenerateJwtToken(AppUser user, string role) {
        var settings = options.Value;
        var expires = DateTime.UtcNow.AddMinutes(settings.ExpirationMin);

        var claims = new List<Claim> {
            new(ClaimTypes.NameIdentifier, user.Id.ToString()),
            new(ClaimTypes.Email, user.Email ?? ""),
            new(ClaimTypes.Role, role),
            new(ClaimTypes.Name, $"{user.Nombre} {user.Apellido}".Trim())
        };

        var key = new SymmetricSecurityKey(Encoding.UTF8.GetBytes(settings.Key));
        var credentials = new SigningCredentials(key, SecurityAlgorithms.HmacSha256);

        var token = new JwtSecurityToken(
            issuer: settings.Issuer,
            audience: settings.Audience,
            claims: claims,
            expires: expires,
            signingCredentials: credentials);

        var tokenString = new JwtSecurityTokenHandler().WriteToken(token);

        return new AuthResponseDto(tokenString, user.Email ?? "", role, expires);
    }

public class FunkoService(
    IFunkoRepository repo,
    IProveedorRepository repoProveedor,
    ICategoriaRepository  repoCategoria,
    ILogger<FunkoService> log,
    IValidator<FunkoCreateRequest> createValidation,
    IValidator<FunkoUpdateRequest> updateValidation,
    IStorageService storage
) : IFunkoService, IScopedService {

    public async Task<PagedResult<FunkoResponseDto>> GetAll(FunkoFilterRequest filter) {
        log.LogInformation("Buscando funkos con filtros {@Filter}", filter);
        var (items, total) = await repo.GetPagedAsync(filter);
        return new PagedResult<FunkoResponseDto> {
            Items = items.Select(f => f.ToDto()).ToList(),
            TotalCount = total,
            Page = filter.Page,
            PageSize = filter.Size
        };
    }

    public async Task<Result<FunkoResponseDto, DomainError>> GetById(int id) {
        log.LogInformation("Obteniendo funko por id: {@Id}", id);
        return (await repo.GetByIdAsync(id)).Map(f => f.ToDto());
    }

    public async Task<Result<FunkoResponseDto, DomainError>> Add(FunkoCreateRequest request) {
        log.LogInformation("Creando funko: {@Funko}", request);
        
        var categoria = await repoCategoria.GetById(request.CategoriaId);
        if (categoria.IsFailure)
            return Result.Failure<FunkoResponseDto, DomainError>(
                new Conflict("La categoría especificada no existe o está eliminada"));

        var proveedor = await repoProveedor.GetById(request.ProveedorId);
        if (proveedor.IsFailure)
            return Result.Failure<FunkoResponseDto, DomainError>(
                new Conflict("El proveedor especificado no existe o está eliminado"));

        var validation = await createValidation.ValidateAsync(request);
        if (!validation.IsValid) {
            log.LogWarning("Validación fallida al crear funko: {@Errors}", validation.Errors);
            return Result.Failure<FunkoResponseDto, DomainError>(
                new ValidationError(validation.Errors[0].ErrorMessage));
        }

        return (await repo.CreateAsync(request.ToModel())).Map(f => f.ToDto());
    }

    public async Task<Result<FunkoResponseDto, DomainError>> Update(int id, FunkoUpdateRequest request) {
        log.LogInformation("Actualizando funko {Id}: {@Funko}", id, request);
        
        var funkoExistente = await repo.GetByIdAsync(id);
        if (funkoExistente.IsFailure)
            return Result.Failure<FunkoResponseDto, DomainError>(funkoExistente.Error);

        if (funkoExistente.Value.IsDeleted)
            return Result.Failure<FunkoResponseDto, DomainError>(
                new Conflict("No se puede actualizar un funko eliminado"));

        var categoria = await repoCategoria.GetById(request.CategoriaId);
        if (categoria.IsFailure)
            return Result.Failure<FunkoResponseDto, DomainError>(
                new Conflict("La categoría especificada no existe o está eliminada"));

        var proveedor = await repoProveedor.GetById(request.ProveedorId);
        if (proveedor.IsFailure)
            return Result.Failure<FunkoResponseDto, DomainError>(
                new Conflict("El proveedor especificado no existe o está eliminado"));

        var validation = await updateValidation.ValidateAsync(request);
        if (!validation.IsValid) {
            log.LogWarning("Validación fallida al actualizar funko {Id}: {@Errors}", id, validation.Errors);
            return Result.Failure<FunkoResponseDto, DomainError>(
                new ValidationError(validation.Errors[0].ErrorMessage));
        }

        return (await repo.UpdateAsync(id, request.ToModel())).Map(f => f.ToDto());
    }

    public async Task<Result<bool, DomainError>> Delete(int id) {
        log.LogInformation("Eliminando funko: {@Id}", id);
        return await repo.DeleteAsync(id);

        storage.SaveFileAsync(file, "funkos");

        if (funko.IsFailure)
            return Result.Failure<FunkoResponseDto, DomainError>(funko.Error);

        var guardada = await storage.SaveFileAsync(file, "funkos");
        if (guardada.IsFailure)
            return Result.Failure<FunkoResponseDto, DomainError>(guardada.Error);

        var actualizado = await repo.SetImagenAsync(id, guardada.Value);
        if (actualizado.IsFailure) {
            var borrado = await storage.DeleteFileAsync(guardada.Value);
            if (borrado.IsFailure)
                log.LogError("No se pudo limpiar la imagen temporal {Imagen}: {Error}",
                    guardada.Value, borrado.Error.Message);
            return Result.Failure<FunkoResponseDto, DomainError>(actualizado.Error);
        }

        if (!string.IsNullOrWhiteSpace(funko.Value.Imagen)) {
            var borrado = await storage.DeleteFileAsync(funko.Value.Imagen);
            if (borrado.IsFailure)
                log.LogWarning("La imagen anterior del funko {Id} no se pudo eliminar: {Error}",
                    id, borrado.Error.Message);
        }

        return actualizado.Map(f => f.ToDto());
    }

using CSharpFunctionalExtensions;
using FunkoApiV._2.Error;
using FunkoApiV._2.Infrastructure.Interfaces;

namespace FunkoApiV._2.Storage;

public class FileSystemStorageService : IStorageService, ITransientService
{
    private readonly string _rootPath;
    private readonly string _uploadPath;
    private readonly long _maxFileSize;
    private readonly HashSet<string> _allowedExtensions;
    private readonly HashSet<string> _allowedContentTypes;
    private readonly ILogger<FileSystemStorageService> _logger;

    public FileSystemStorageService(IConfiguration configuration, ILogger<FileSystemStorageService> logger, IWebHostEnvironment env)
    {
        _logger = logger;

        var configuredUploadPath = (configuration["Storage:UploadPath"] ?? "uploads")
            .Replace('\\', '/')
            .Trim('/');
        _uploadPath = configuredUploadPath.StartsWith("wwwroot/", StringComparison.OrdinalIgnoreCase)
            ? configuredUploadPath["wwwroot/".Length..]
            : configuredUploadPath;
        _maxFileSize = configuration.GetValue<long>("Storage:MaxFileSize", 5 * 1024 * 1024);
        _allowedExtensions = (configuration.GetSection("Storage:AllowedExtensions").Get<string[]>()
            ?? [".jpg", ".jpeg", ".png", ".gif"])
            .ToHashSet(StringComparer.OrdinalIgnoreCase);
        _allowedContentTypes = (configuration.GetSection("Storage:AllowedContentTypes").Get<string[]>()
            ?? ["image/jpeg", "image/png", "image/gif"])
            .ToHashSet(StringComparer.OrdinalIgnoreCase);

        var webRoot = env.WebRootPath ?? System.IO.Path.Combine(env.ContentRootPath, "wwwroot");
        _rootPath = System.IO.Path.GetFullPath(System.IO.Path.Combine(webRoot, _uploadPath));

        if (!Directory.Exists(_rootPath))
            Directory.CreateDirectory(_rootPath);

        _logger.LogInformation("Storage service inicializado en: {Path}", _rootPath);
    }

    private DomainError? ValidateFile(IFormFile file) {
        if (file is null or { Length: 0 })
            return StorageError.ArchivoVacio();
        if (file.Length > _maxFileSize)
            return StorageError.ArchivoMuyGrande();
        if (string.IsNullOrWhiteSpace(file.FileName))
            return StorageError.NombreArchivoInvalido();

        var extension = System.IO.Path.GetExtension(file.FileName);
        if (!_allowedExtensions.Contains(extension))
            return StorageError.ExtensionNoPermitida();
        if (string.IsNullOrWhiteSpace(file.ContentType) ||
            !_allowedContentTypes.Contains(file.ContentType))
            return StorageError.TipoContenidoNoPermitido();

        return null;
    }

    public async Task<Result<string, DomainError>> SaveFileAsync(IFormFile file, string folder) {
        var validationError = ValidateFile(file);
        if (validationError is not null)
            return Result.Failure<string, DomainError>(validationError);

        string? filePath = null;
        try
        {
            var extension = System.IO.Path.GetExtension(file.FileName).ToLowerInvariant();
            var filename = $"{Guid.NewGuid():N}{extension}";
            var relativePath = GetRelativePath(filename, folder);
            filePath = GetFullPath(relativePath);
            var folderPath = System.IO.Path.GetDirectoryName(filePath)!;
            Directory.CreateDirectory(folderPath);

            await using var stream = new FileStream(
                filePath,
                FileMode.CreateNew,
                FileAccess.Write,
                FileShare.None,
                81920,
                useAsync: true);
            await file.CopyToAsync(stream);

            _logger.LogInformation("Archivo guardado: {Path}", relativePath);
            return Result.Success<string, DomainError>(relativePath);
        }
        catch (Exception ex)
        {
            if (filePath is not null && File.Exists(filePath)) {
                try {
                    File.Delete(filePath);
                }
                catch (Exception cleanupException) {
                    _logger.LogError(cleanupException, "No se pudo limpiar el archivo incompleto {Path}", filePath);
                }
            }
            _logger.LogError(ex, "Error guardando archivo");
            return Result.Failure<string, DomainError>(StorageError.ErrorGuardando());
        }
    }

    public Task<Result<bool, DomainError>> DeleteFileAsync(string relativePath) {
        if (string.IsNullOrWhiteSpace(relativePath))
            return Task.FromResult(Result.Success<bool, DomainError>(true));

        try
        {
            var fullPath = GetFullPath(relativePath);
            if (File.Exists(fullPath))
            {
                File.Delete(fullPath);
                _logger.LogInformation("Archivo eliminado: {Filename}", relativePath);
            }
            return Task.FromResult(Result.Success<bool, DomainError>(true));
        }
        catch (Exception ex)
        {
            _logger.LogError(ex, "Error eliminando archivo {Filename}", relativePath);
            return Task.FromResult(Result.Failure<bool, DomainError>(StorageError.ErrorEliminando()));
        }
    }

    private string GetFullPath(string filename) {
        var relativePath = filename.Replace('\\', '/').TrimStart('/');
        var uploadPrefix = $"{_uploadPath.TrimEnd('/')}/";
        if (relativePath.StartsWith(uploadPrefix, StringComparison.OrdinalIgnoreCase))
            relativePath = relativePath[uploadPrefix.Length..];

        var fullPath = System.IO.Path.GetFullPath(System.IO.Path.Combine(
            _rootPath,
            relativePath.Replace('/', System.IO.Path.DirectorySeparatorChar)));
        var rootPrefix = _rootPath.TrimEnd(
            System.IO.Path.DirectorySeparatorChar,
            System.IO.Path.AltDirectorySeparatorChar) + System.IO.Path.DirectorySeparatorChar;
        if (!fullPath.StartsWith(rootPrefix, StringComparison.OrdinalIgnoreCase))
            throw new ArgumentException("La ruta debe estar dentro del almacenamiento configurado.", nameof(filename));

        return fullPath;
    }

    private string GetRelativePath(string filename, string folder) {
        var normalizedFolder = folder.Replace('\\', '/').Trim('/');
        if (string.IsNullOrWhiteSpace(normalizedFolder) ||
            normalizedFolder.Split('/').Any(part => part is "." or "..")) {
            throw new ArgumentException("La carpeta de almacenamiento no es vÃ¡lida.", nameof(folder));
        }

        return $"/{_uploadPath}/{normalizedFolder}/{filename}";
    }
}
